        """
        주간 성장률 계산
        
        Returns:
            float: (이번주 - 전주) / 전주 * 100

    """통합 운영 현황 응답"""

    riskUsers: List[RiskUserDto]            # 위험 유저 상세 리스트
    opportunityUsers: List[OpportunityUserDto]  # 기회 유저 상세 리스트


class RevenueStatsDto(BaseModel):
    """수익 통계 (Game Log 기반)"""
    today_revenue: int
    today_expenses: int
    net_income: int
    deposit_count: int
    weekly_growth_rate: float


class RiskUserDto(BaseModel):
    """위험 유저 상세"""
    user_id: int
    nickname: str
    risk_type: str  # "LOSS_STREAK" | "BALANCE_DROP" | "INACTIVE"
    risk_level: str  # "HIGH" | "MEDIUM"
    risk_score: float
    details: dict  # 구체적 수치 (loss_streak=7, balance_drop_pct=60 등)
    last_activity_at: datetime


class OpportunityUserDto(BaseModel):

# app/v2/models/v2_game_log.py

class V2GameLog(Base):
    """외부 게임 로그 저장 테이블"""
    __tablename__ = "v2_game_log"
    
    id = Column(BigInteger, primary_key=True, autoincrement=True)
    user_id = Column(Integer, ForeignKey("v2_users.id"), nullable=False, index=True)
    game_type = Column(String(32), nullable=False)  # DICE, ROULETTE, SLOT, POKER
    result = Column(String(16), nullable=False)     # WIN, LOSE, DRAW, JACKPOT
    bet_amount = Column(BigInteger, nullable=False)
    payout_amount = Column(BigInteger, nullable=False)
    balance_after = Column(BigInteger, nullable=False)
    external_user_id = Column(String(128), nullable=True)
    session_id = Column(String(128), nullable=True)
    game_metadata = Column(JSON, nullable=True)
    recorded_at = Column(DateTime(timezone=True), nullable=False)  # CSV의 기록 일시
    imported_at = Column(DateTime(timezone=True), nullable=False, server_default=func.now())
    
    __table_args__ = (
        Index("ix_v2_game_log_user_recorded", "user_id", "recorded_at"),
        Index("ix_v2_game_log_recorded_at", "recorded_at"),
    )

## 5. 데이터 파이프라인 통합

### 5.1 CSV Import → DB 저장

```
Game Log CSV
    ↓
CSVImportService.import_csv()
    ↓
├── Redis Pub/Sub (실시간 분석용)
│
└── V2GameLog 테이블 저장 (영구 보관)
```

### 5.2 V2GameLog → Analytics

```
V2GameLog 테이블
    ↓
GameLogAnalyticsService
    ↓
├── get_today_revenue() → AnalyticsDashboard 수익/지출
├── get_weekly_growth_rate() → AnalyticsDashboard 성장률
├── get_high_rollers() → OpsDashboard 기회 그룹
└── detect_loss_streak_users() → OpsDashboard 위험 그룹
```

### 5.3 HQ Margin → Ops Dashboard

```
V2UserSegment + HQProspectiveUser
    ↓
HQMarginStatsService (기존)
    ↓
└── 통합: 위험군(AT_RISK) + 기회군(VIP/WHALE) 상세 리스트
```
